    const CLEAN_REDIRECT_URI = window.location.origin + window.location.pathname;

    function generateRandomString(length) {

      const possible = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789';

      const values = crypto.getRandomValues(new Uint8Array(length));

      return values.reduce((acc, x) => acc + possible[x % possible.length], "");

    }

    async function sha256(plain) {

      const encoder = new TextEncoder();

      const data = encoder.encode(plain);

      return window.crypto.subtle.digest('SHA-256', data);

    }

    function base64encode(input) {

      return btoa(String.fromCharCode(...new Uint8Array(input)))

        .replace(/=/g, '')

        .replace(/\+/g, '-')

        .replace(/\//g, '_');

    }

    async function startSpotifyAuth() {

      if (!state.clientId) {

        openSettings();

        return;

      }

      const codeVerifier = generateRandomString(64);

      localStorage.setItem('sp_code_verifier', codeVerifier);

      const hashed = await sha256(codeVerifier);

      const codeChallenge = base64encode(hashed);

      const scope = [

        'user-read-playback-state',

        'user-modify-playback-state',

        'user-read-currently-playing',

        'user-library-read',

        'user-library-modify',

        'streaming'

      ].join(' ');

      const authUrl = new URL("https://accounts.spotify.com/authorize");

      authUrl.search = new URLSearchParams({

        response_type: 'code',

        client_id: state.clientId,

        scope,

        code_challenge_method: 'S256',

        code_challenge: codeChallenge,

        redirect_uri: CLEAN_REDIRECT_URI,

      }).toString();

      window.location.href = authUrl.toString();

    }

    async function handleAuthCallback() {

      const params = new URLSearchParams(window.location.search);

      const code = params.get('code');

      if (!code) return;

      const codeVerifier = localStorage.getItem('sp_code_verifier');

      if (!codeVerifier || !state.clientId) return;

      try {

        const res = await fetch('https://accounts.spotify.com/api/token', {

          method: 'POST',

          headers: { 'Content-Type': 'application/x-www-form-urlencoded' },

          body: new URLSearchParams({

            client_id: state.clientId,

            grant_type: 'authorization_code',

            code,

            redirect_uri: CLEAN_REDIRECT_URI,

            code_verifier: codeVerifier,

          }),

        });

        const data = await res.json();

        if (data.access_token) {

          saveTokens(data);

          window.history.replaceState({}, document.title, CLEAN_REDIRECT_URI);

        }

      } catch (e) {

        console.error('Auth error:', e);

      }

    }

    function saveTokens(data) {

      state.accessToken = data.access_token;

      if (data.refresh_token) state.refreshToken = data.refresh_token;

      state.tokenExpiry = Date.now() + (data.expires_in - 60) * 1000;

      localStorage.setItem('sp_access_token', state.accessToken);

      if (state.refreshToken) localStorage.setItem('sp_refresh_token', state.refreshToken);

      localStorage.setItem('sp_token_expiry', String(state.tokenExpiry));

    }

    async function getValidToken() {

      if (!state.accessToken) return null;

      if (Date.now() < state.tokenExpiry) return state.accessToken;

      if (!state.refreshToken || !state.clientId) return state.accessToken;

      try {

        const res = await fetch('https://accounts.spotify.com/api/token', {

          method: 'POST',

          headers: { 'Content-Type': 'application/x-www-form-urlencoded' },

          body: new URLSearchParams({

            client_id: state.clientId,

            grant_type: 'refresh_token',

            refresh_token: state.refreshToken

          }),

        });

        const data = await res.json();

        if (data.access_token) {

          saveTokens(data);

          return state.accessToken;

        }

      } catch (e) {

        console.error('Refresh error:', e);

      }

      return state.accessToken;

    }

    async function spotifyApi(endpoint, method = 'GET', body = null) {

      const token = await getValidToken();

      if (!token) return null;

      const options = {

        method,

        headers: {

          'Authorization': `Bearer ${token}`,

          'Content-Type': 'application/json'

        }

      };

      if (body) options.body = JSON.stringify(body);

      const res = await fetch(`https://api.spotify.com/v1${endpoint}`, options);

      if (res.status === 204) return { ok: true };

      if (!res.ok) {

        return null;

      }

      return res.json();

    }

    function pickRandom(arr, count) {

      const copy = [...arr];

      for (let i = copy.length - 1; i > 0; i--) {

        const j = Math.floor(Math.random() * (i + 1));

        [copy[i], copy[j]] = [copy[j], copy[i]];

      }

      return copy.slice(0, count);

    }

    function formatSpotifyTrack(item, reason = '') {

      return {

        uri: item.uri,

        id: item.id,

        name: item.name,

        artist: item.artists.map(a => a.name).join(', '),

        album: item.album?.name || '',

        cover: item.album?.images?.[1]?.url || item.album?.images?.[0]?.url || '',

        durationMs: item.duration_ms || 0,

        reason

      };

    }

    async function searchSpotifyTracks(query, limit = 5, randomizeOffset = true) {

      const offset = randomizeOffset ? Math.floor(Math.random() * 15) : 0;

      const q = encodeURIComponent(query);

      let data = await spotifyApi(`/search?q=${q}&type=track&limit=${limit}&offset=${offset}`);

      if (!data?.tracks?.items?.length && offset > 0) {

        data = await spotifyApi(`/search?q=${q}&type=track&limit=${limit}&offset=0`);

      }

      return data?.tracks?.items || [];

    }

    async function fetchArtistTracks(artistName, count = 3, exactTop = false) {

      const items = await searchSpotifyTracks(`artist:"${artistName}"`, 10, !exactTop);

      const filtered = items.filter(t =>

        t.artists.some(a => a.name.toLowerCase().includes(artistName.toLowerCase()))

      );

      const source = filtered.length ? filtered : items;

      return pickRandom(source, count).map(t => formatSpotifyTrack(t, `По запросу: ${artistName}`));

    }

    async function generateFreshBatch(count = 5) {

      if (!state.accessToken) {

        return getFallbackMockBatch(count);

      }

      let pool = USER_TASTE_POOL;

      let tag = 'Твой профиль';

      if (state.vibeMode === 'work') {

        pool = WORK_FOCUS_ARTISTS;

        tag = 'Фокус и работа';

      } else if (state.vibeMode === 'heavy') {

        pool = HEAVY_ARTISTS;

        tag = 'Тяжёлый драйв';

      }

      const chosenArtists = pickRandom(pool, count + 2);

      const results = [];

      const usedUris = new Set([

        state.currentTrack?.uri,

        ...state.queue.map(t => t.uri),

        ...state.history.slice(0, 15).map(t => t.uri)

      ]);

      for (const artist of chosenArtists) {

        if (results.length >= count) break;

        const items = await searchSpotifyTracks(`artist:"${artist}"`, 8, true);

        const candidates = items.filter(t => !usedUris.has(t.uri));

        if (candidates.length > 0) {

          const picked = candidates[Math.floor(Math.random() * candidates.length)];

          usedUris.add(picked.uri);

          results.push(formatSpotifyTrack(picked, `${tag} · ${artist}`));

        }

      }

      return results;

    }

      ].slice(0, count);

    }

    function addToHistory(track) {

      if (!track || !track.uri || track.uri.startsWith('mock:')) return;

      if (state.history[0]?.uri === track.uri) return;

      const now = new Date();

      const timeStr = `${String(now.getHours()).padStart(2, '0')}:${String(now.getMinutes()).padStart(2, '0')}`;

      state.history.unshift({ ...track, playedAt: timeStr });

      if (state.history.length > 30) state.history.pop();

      renderHistory();

    }

    async function refillQueue(forceNew = false) {

      if (forceNew) {

        state.queue = await generateFreshBatch(5);

        renderQueue();

        if (state.djActive && state.currentTrack) {

          await playQueueOnSpotify([state.currentTrack, ...state.queue]);

        }

        return;

      }

      const needed = 5 - state.queue.length;

      if (needed > 0) {

        const more = await generateFreshBatch(needed);

        state.queue.push(...more);

      }

      state.queue = state.queue.slice(0, 5);

      renderQueue();

    }

    async function syncPlaybackState() {

      if (!state.accessToken) return;

      const data = await spotifyApi('/me/player');

      if (!data || !data.item) return;

      state.isPlaying = Boolean(data.is_playing);

      updatePlayPauseUI();

      const liveTrack = formatSpotifyTrack(data.item, state.currentTrack?.uri === data.item.uri ? state.currentTrack.reason : 'Spotify Connect');

      if (state.lastTrackUri && state.lastTrackUri !== liveTrack.uri) {

        const qIdx = state.queue.findIndex(t => t.uri === liveTrack.uri);

        if (qIdx !== -1) {

          state.queue.splice(qIdx, 1);

          await refillQueue(false);

        }

        setCurrentTrack(liveTrack);

      } else if (!state.currentTrack) {

        setCurrentTrack(liveTrack);

      }

      const progress = data.progress_ms || 0;

      const duration = data.item.duration_ms || 1;

      const pct = Math.min(100, Math.round((progress / duration) * 100));

      document.getElementById('progressBar').style.width = `${pct}%`;

      document.getElementById('timeElapsed').textContent = formatTime(progress);

      document.getElementById('timeTotal').textContent = formatTime(duration);

    }

    async function processCommandLocally(text) {

      const lower = text.toLowerCase();

      const countArtistMatch = text.match(/(?:поставь|включи|добавь|дай|сыграй)\s+(\d+)\s+(?:трек\w*|пес\w*)\s+(?:группы|исполнителя|артиста)?\s*(.+)/i);

      if (countArtistMatch) {

        const count = Math.min(5, Math.max(1, parseInt(countArtistMatch[1], 10)));

        const artistQuery = countArtistMatch[2].replace(/[?.!]+$/, '').trim();

        addChatMessage('dj', `Ищу ${count} трек(а) ${artistQuery} в Spotify...`);

        const tracks = await fetchArtistTracks(artistQuery, count, false);

        if (!tracks.length) {

          addChatMessage('dj', `Не нашёл треки «${artistQuery}» в Spotify. Проверь написание названия.`);

          return;

        }

        const filler = await generateFreshBatch(Math.max(0, 5 - tracks.length));

        const newFirst = tracks[0];

        state.queue = [...tracks.slice(1), ...filler].slice(0, 5);

        setCurrentTrack(newFirst);

        renderQueue();

        state.djActive = true;

        updateDjButtonUI();

        await playQueueOnSpotify([newFirst, ...state.queue]);

        addChatMessage('dj', `Поставил ${tracks.length} трек(а) ${newFirst.artist}. Первым играет «${newFirst.name}».`);

        return;

      }

      const specificTrackMatch = text.match(/(?:включи|поставь|найди|сыграй)\s+(?:трек|песню)\s+(.+)/i);

      if (specificTrackMatch) {

        const query = specificTrackMatch[1].replace(/[?.!]+$/, '').trim();

        const items = await searchSpotifyTracks(query, 3, false);

        if (!items.length) {

          addChatMessage('dj', `Трек «${query}» не найден.`);

          return;

        }

        const chosen = formatSpotifyTrack(items[0], 'По твоему запросу');

        setCurrentTrack(chosen);

        await refillQueue(false);

        state.djActive = true;

        updateDjButtonUI();

        await playQueueOnSpotify([chosen, ...state.queue]);

        addChatMessage('dj', `Запускаю: ${chosen.artist} — «${chosen.name}».`);

        return;

      }

      if (lower.includes('спокойн') || lower.includes('работ') || lower.includes('фокус') || lower.includes('чилл') || lower.includes('инструментал')) {

        state.vibeMode = 'work';

        document.getElementById('djVibeSummary').textContent = 'Фокус и работа · Инструментал · Инди';

        await refillQueue(true);

        if (state.queue.length > 0) await playFromQueueIndex(0);

        addChatMessage('dj', 'Переключил эфир на спокойный рабочий режим: kensuke ushio, toe, the pillows, RADWIMPS.');

        return;

      }

      if (lower.includes('тяжел') || lower.includes('тяжёл') || lower.includes('метал') || lower.includes('драйв') || lower.includes('бодр')) {

        state.vibeMode = 'heavy';

        document.getElementById('djVibeSummary').textContent = 'Тяжёлый драйв · Пост-хардкор · J-Rock';

        await refillQueue(true);

        if (state.queue.length > 0) await playFromQueueIndex(0);

        addChatMessage('dj', 'Добавил плотного звука и драйва: Ling tosite sigure, Bring Me The Horizon, MAXIMUM THE HORMONE.');

        return;

      }

      const cleanedQuery = text.replace(/^(поставь|включи|дай|хочу|добавь|что-нибудь|похожее на|в духе)\s+/i, '').replace(/[?.!]+$/, '').trim();

      if (cleanedQuery.length >= 2) {

        const items = await searchSpotifyTracks(cleanedQuery, 8, true);

        if (items.length > 0) {

          const formatted = pickRandom(items, 3).map(t => formatSpotifyTrack(t, `Запрос: ${cleanedQuery}`));

          const filler = await generateFreshBatch(5 - formatted.length);

          const first = formatted[0];

          state.queue = [...formatted.slice(1), ...filler].slice(0, 5);

          setCurrentTrack(first);

          renderQueue();

          state.djActive = true;

          updateDjButtonUI();

          await playQueueOnSpotify([first, ...state.queue]);

          addChatMessage('dj', `Обновил сет по запросу «${cleanedQuery}». Сейчас играет ${first.artist} — «${first.name}».`);

          return;

        }

      }

      addChatMessage('dj', 'Не совсем понял запрос. Попробуй написать так: «поставь 3 трека Ado», «сделай спокойнее для работы» или «включи трек Otonoke».');

    }

    async function processCommandWithGemini(userText) {

      try {

        const prompt = `Ты AI DJ в Spotify. Вкус пользователя: Ado, Creepy Nuts, MASS OF THE FERMENTING DREGS, the pillows, Ling tosite sigure, J-Rock, математический рок, японское инди, металкор.

Пользователь написал: "${userText}".

Верни ТОЛЬКО валидный JSON без markdown-обёртки в формате:

{"reply": "короткий ответ диджея на русском", "queries": ["поисковый запрос 1 для Spotify", "запрос 2", "запрос 3"], "vibe": "balanced|work|heavy"}

Если пользователь просит N треков конкретной группы, укажи в queries N раз название этой группы (например ["artist:Ling tosite sigure", "artist:Ling tosite sigure", "artist:Ling tosite sigure"]).`;

        const res = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${state.geminiKey}`, {

          method: 'POST',

          headers: { 'Content-Type': 'application/json' },

          body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] })

        });

        const data = await res.json();

        const raw = data?.candidates?.[0]?.content?.parts?.[0]?.text || '';

        const jsonStr = raw.replace(/```json|```/g, '').trim();

        const parsed = JSON.parse(jsonStr);

        if (parsed.vibe) state.vibeMode = parsed.vibe;

        const foundTracks = [];

        const used = new Set();

        for (const q of (parsed.queries || [])) {

          const items = await searchSpotifyTracks(q, 6, true);

          const fresh = items.find(t => !used.has(t.uri));

          if (fresh) {

            used.add(fresh.uri);

            foundTracks.push(formatSpotifyTrack(fresh, 'AI DJ'));

          }

        }

        if (foundTracks.length > 0) {

          const filler = await generateFreshBatch(Math.max(0, 5 - foundTracks.length));

          const first = foundTracks[0];

          state.queue = [...foundTracks.slice(1), ...filler].slice(0, 5);

          setCurrentTrack(first);

          renderQueue();

          state.djActive = true;

          updateDjButtonUI();

          await playQueueOnSpotify([first, ...state.queue]);

          addChatMessage('dj', parsed.reply || `Запускаю ${first.artist} — «${first.name}».`);

          return true;

        }

      } catch (e) {

        console.warn('Gemini fallback to local parser:', e);

      }

      return false;

    }

      input.value = '';

        try {

          const ctx = state.currentTrack ? `Текущий трек: ${state.currentTrack.artist} — ${state.currentTrack.name} (альбом: ${state.currentTrack.album}). ` : '';

          const res = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${state.geminiKey}`, {

            method: 'POST',

            headers: { 'Content-Type': 'application/json' },

            body: JSON.stringify({

              contents: [{ parts: [{ text: `${ctx}Ответь кратко, по делу и на русском языке на вопрос пользователя: ${question}` }] }]

            })

          });

          const data = await res.json();

          if (answer) {

            addInfoMessage('ai', answer);

            return;

    function escapeHtml(str) {

      return String(str || '').replace(/[&<>"']/g, m => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[m]));

    }
